function setError(field, on){ field.classList.toggle("error", on); return !on; }
function validateForm(form){
let ok = true;
form.querySelectorAll(".field").forEach(f=>f.classList.remove("error"));
const name = form.querySelector('[name="name"]');
if(name && name.value.trim().length < 2){ setError(name.closest(".field"), true); ok = false; }
const phone = form.querySelector('[name="phone"]');
if(phone){ if(phone.value.replace(/\D/g,"").length < 10){ setError(phone.closest(".field"), true); ok = false; } }
const email = form.querySelector('[name="email"]');
if(email && email.required){ if(!/^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(email.value)){ setError(email.closest(".field"), true); ok = false; } }
return ok;
}

async function sendViaFormSubmit(email, payload){
if(!email || email === "you@example.com") return null;
try {
const res = await fetch("https://formsubmit.co/ajax/" + email, {
method: "POST", headers: { "Content-Type": "application/json", "Accept": "application/json" }, body: JSON.stringify(payload)
});
return await res.json();
} catch(e){ console.warn("FormSubmit:", e); return null; }
}
